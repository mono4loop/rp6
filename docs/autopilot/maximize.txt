# Live maximize check, run by `make smoke-maximize` (scripts/smoke-maximize.sh
# also checks the RP6_DIAG log it leaves). The window commands make the same
# compositor requests the title bar does; the console shows the keyboard, the
# window layout puts it away again.
wait idle
window maximize
wait maximized 5s
wait keyboard_keys 5s
window restore
wait windowed 5s
wait gone keyboard_keys 5s
wait idle
quit
